#6.

Loops - block of code used repeatly until condition is true.

for = Execute Specific number of times.

for(let i=0;i<5;i++){
    console.log(i);
}
    
while  = Execute condition until true.

let i=0;
while(i<5){
    console.log(i);
    i++;
}

do-while = Execute atleast once.

let i=0;
do{
    console.log(i);
    i++;
}while(i<4);
