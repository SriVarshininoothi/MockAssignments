2. useState is a React hook used to store state.It is used to create and manage inside componenet.

import React, { useState } from 'react';

const Counter = () =>{

    //store count
const[count,setCount]=useState(0);

//handleIncrement

const handleIncrement = ()=>{
    setCount(count+1);

}

return(
    <div>
        <button onClick = {handleIncrement}>Add</button>
    </div>
)

}
export default Counter;
